# "clean_jni": the robot driver with the C API and its JNI bridge, for the Kotlin apps
# (Android through the NDK, desktop through :shared:buildCoreJni).
# CLEAN_DRIVER: the folder in drivers/ to build (needed when there is more than one).
cmake_minimum_required(VERSION 3.15)
project(clean_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(CLEAN_DRIVER "" CACHE STRING "robot driver (a folder in drivers/)")
if(CLEAN_DRIVER STREQUAL "")
  file(GLOB FOUND ${CMAKE_CURRENT_LIST_DIR}/../drivers/*/src/clean_c.cpp)   # the scraper drivers
  list(LENGTH FOUND N)
  if(NOT N EQUAL 1)
    message(FATAL_ERROR "set CLEAN_DRIVER to a folder in drivers/")
  endif()
  get_filename_component(FOUND ${FOUND} DIRECTORY)
  get_filename_component(FOUND ${FOUND} DIRECTORY)
  get_filename_component(CLEAN_DRIVER ${FOUND} NAME)
endif()
get_filename_component(DRIVER_SRC ${CMAKE_CURRENT_LIST_DIR}/../drivers/${CLEAN_DRIVER}/src ABSOLUTE)
file(GLOB DRIVER_SOURCES ${DRIVER_SRC}/*.cpp)
set(JNI_SOURCES ${CMAKE_CURRENT_LIST_DIR}/jni/clean_jni.cpp)
if(NOT ANDROID)   # classic Bluetooth serial for the desktop app (Android has its own API)
  if(WIN32)
    set(BT_OS windows)
    set(BT_EXT .cpp)
  elseif(APPLE)
    set(BT_OS macos)
    set(BT_EXT .mm)   # IOBluetooth is Objective-C
    enable_language(OBJCXX)
  elseif(UNIX)
    set(BT_OS linux)
    set(BT_EXT .cpp)
  else()
    set(BT_OS none)
    set(BT_EXT .cpp)
  endif()
  list(APPEND JNI_SOURCES ${CMAKE_CURRENT_LIST_DIR}/jni/bt_jni.cpp ${CMAKE_CURRENT_LIST_DIR}/jni/bt_${BT_OS}${BT_EXT})
endif()
# Updates of the desktop app on Windows and macOS: Velopack's C library, linked in
# (jni/updates_jni.cpp). Linux updates through apt (.deb).
if(NOT ANDROID AND (WIN32 OR APPLE))
  set(VELOPACK ON)
  include(FetchContent)
  FetchContent_Declare(velopack
    URL https://github.com/velopack/velopack/releases/download/1.2.161/velopack_libc_1.2.161.zip
    URL_HASH SHA256=31de8f2cf05a47bd5328a9f98b0c41aec1582156f6075f1058b26941f7f17c20
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(velopack)   # headers and libraries only (no CMakeLists.txt)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
    set(VPK_ARCH arm64)
  else()
    set(VPK_ARCH x64)
  endif()
  if(WIN32)
    set(VPK_LIB ${velopack_SOURCE_DIR}/lib-static/velopack_libc_win_${VPK_ARCH}_msvc.lib)
  else()
    set(VPK_LIB ${velopack_SOURCE_DIR}/lib-static/velopack_libc_osx_${VPK_ARCH}_gnu.a)
  endif()
  list(APPEND JNI_SOURCES ${CMAKE_CURRENT_LIST_DIR}/jni/updates_jni.cpp)
endif()
add_library(clean_jni SHARED ${DRIVER_SOURCES} ${JNI_SOURCES})
target_include_directories(clean_jni PRIVATE ${DRIVER_SRC} ${CMAKE_CURRENT_LIST_DIR})
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(clean_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
find_package(Threads REQUIRED)
target_link_libraries(clean_jni PRIVATE Threads::Threads)
if(ANDROID)
  target_link_options(clean_jni PRIVATE "-Wl,-z,max-page-size=16384")
elseif(UNIX AND NOT APPLE)
  # one self-contained .so for the desktop app
  target_link_options(clean_jni PRIVATE -static-libstdc++ -static-libgcc)
elseif(MSVC)
  # the same on Windows: no separate C++ runtime to install
  set_property(TARGET clean_jni PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
if(WIN32)
  target_link_libraries(clean_jni PRIVATE ws2_32)
endif()
if(VELOPACK)
  target_include_directories(clean_jni PRIVATE ${velopack_SOURCE_DIR}/include)
  target_link_libraries(clean_jni PRIVATE ${VPK_LIB})
  if(WIN32)
    target_link_libraries(clean_jni PRIVATE userenv bcrypt ntdll advapi32 crypt32 secur32 ole32 shell32 oleaut32 propsys)
  else()
    target_link_libraries(clean_jni PRIVATE "-framework CoreFoundation" "-framework Security" "-framework SystemConfiguration" "-framework AppKit")
  endif()
endif()
if(APPLE AND NOT ANDROID)
  set_source_files_properties(${CMAKE_CURRENT_LIST_DIR}/jni/bt_macos.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(clean_jni PRIVATE "-framework IOBluetooth" "-framework Foundation")
endif()
